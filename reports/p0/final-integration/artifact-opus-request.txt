#!/usr/bin/env python3
"""Point-in-time diagnostic artifact tree digests, not execution authorization.

This tool never makes source or data immutable, authenticates a manifest, or
proves that a job actually reads these bytes. Concurrent writers and changes
between verification and use require a separate immutable execution boundary.
"""
import hashlib
import os
from pathlib import Path
import stat


class ArtifactError(Exception):
    pass


def _identity(info):
    return (info.st_dev, info.st_ino, info.st_mode, info.st_size,
            info.st_mtime_ns, info.st_ctime_ns)


def snapshot(root, max_entries=100000):
    """Hash a directory without following child symlinks or opening devices."""
    if not isinstance(max_entries, int) or isinstance(max_entries, bool) or max_entries < 1:
        raise ArtifactError('invalid entry limit')
    records = []
    def walk(fd, prefix):
        before = os.fstat(fd)
        for name in sorted(os.listdir(fd)):
            if len(records) >= max_entries:
                raise ArtifactError('entry limit exceeded')
            rel = prefix + name
            info = os.stat(name, dir_fd=fd, follow_symlinks=False)
            if stat.S_ISDIR(info.st_mode):
                child = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
                try:
                    if (info.st_dev, info.st_ino) != (os.fstat(child).st_dev, os.fstat(child).st_ino):
                        raise ArtifactError('directory changed during scan')
                    records.append({'path': rel, 'type': 'directory'})
                    walk(child, rel + '/')
                finally:
                    os.close(child)
            elif stat.S_ISREG(info.st_mode):
                # O_PATH pins the inode without opening a swapped device/FIFO.
                # Reopen the pinned regular inode via procfs, then verify identity.
                if not hasattr(os, 'O_PATH'):
                    raise ArtifactError('Linux O_PATH support required')
                handle = os.open(name, os.O_PATH | os.O_NOFOLLOW, dir_fd=fd)
                try:
                    pinned = os.fstat(handle)
                    if not stat.S_ISREG(pinned.st_mode) or _identity(info) != _identity(pinned):
                        raise ArtifactError('file changed before read')
                    child = os.open('/proc/self/fd/%d' % handle, os.O_RDONLY | os.O_NONBLOCK)
                    try:
                        opened = os.fstat(child)
                        if _identity(pinned) != _identity(opened):
                            raise ArtifactError('file changed before read')
                        digest = hashlib.sha256()
                        while True:
                            chunk = os.read(child, 1024 * 1024)
                            if not chunk:
                                break
                            digest.update(chunk)
                        if _identity(opened) != _identity(os.fstat(child)):
                            raise ArtifactError('file changed during read')
                        records.append({'path': rel, 'type': 'file', 'size': opened.st_size,
                                        'sha256': digest.hexdigest()})
                    finally:
                        os.close(child)
                finally:
                    os.close(handle)
            else:
                raise ArtifactError('symlink or non-regular artifact is not allowed')
        if _identity(before) != _identity(os.fstat(fd)):
            raise ArtifactError('directory changed during scan')
    fd = None
    try:
        fd = os.open(Path(root), os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        walk(fd, '')
        return {'schema_version': 1, 'entries': records}
    except OSError as exc:
        raise ArtifactError('cannot safely scan artifact tree') from exc
    finally:
        if fd is not None:
            os.close(fd)


def verify(root, manifest):
    """Compare actual files to a trusted supplied manifest; does not verify trust."""
    return snapshot(root) == manifest


def main():
    import argparse
    import json
    import sys
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('root')
    parser.add_argument('--verify', type=Path, help='Compare against a previously saved JSON manifest')
    args = parser.parse_args()
    try:
        actual = snapshot(args.root)
        if args.verify:
            expected = json.loads(args.verify.read_text(encoding='utf-8'))
            if actual != expected:
                print('Artifact tree differs', file=sys.stderr)
                return 1
        print(json.dumps(actual, ensure_ascii=True, sort_keys=True, indent=2))
        return 0
    except (ArtifactError, OSError, ValueError) as exc:
        print('BLOCKED: %s' % exc, file=sys.stderr)
        return 2


if __name__ == '__main__':
    raise SystemExit(main())

"""CPU-only integrity checks for diagnostic artifact tree manifests."""
import importlib.util
import pathlib
import tempfile
import unittest

spec = importlib.util.spec_from_file_location('artifact_tree', pathlib.Path(__file__).resolve().parents[1] / 'scripts' / 'artifact_tree.py')
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)

class ArtifactTreeTests(unittest.TestCase):
    def test_changes_additions_and_removals_detected(self):
        with tempfile.TemporaryDirectory() as tmp:
            root = pathlib.Path(tmp)
            (root / 'data').mkdir()
            (root / 'data' / 'x').write_bytes(b'first')
            record = m.snapshot(root)
            self.assertTrue(m.verify(root, record))
            (root / 'data' / 'x').write_bytes(b'other')
            self.assertFalse(m.verify(root, record))
            (root / 'data' / 'x').write_bytes(b'first')
            (root / 'other').write_bytes(b'new')
            self.assertFalse(m.verify(root, record))
            (root / 'other').unlink()
            (root / 'data' / 'x').unlink()
            self.assertFalse(m.verify(root, record))

    def test_symlink_and_fifo_fail_closed(self):
        import os
        with tempfile.TemporaryDirectory() as tmp:
            root = pathlib.Path(tmp)
            (root / 'link').symlink_to('/etc/passwd')
            with self.assertRaises(m.ArtifactError): m.snapshot(root)
            (root / 'link').unlink()
            os.mkfifo(root / 'pipe')
            with self.assertRaises(m.ArtifactError): m.snapshot(root)

    def test_manifest_is_deterministic_and_detects_empty_directory(self):
        with tempfile.TemporaryDirectory() as tmp:
            root = pathlib.Path(tmp)
            (root / 'z').write_bytes(b'z')
            (root / 'a').mkdir()
            first = m.snapshot(root)
            self.assertEqual(first, m.snapshot(root))
            (root / 'a').rmdir()
            self.assertFalse(m.verify(root, first))

    def test_command_verifies_actual_files(self):
        import json
        import subprocess
        import sys
        with tempfile.TemporaryDirectory() as tmp:
            root = pathlib.Path(tmp) / 'data'
            root.mkdir()
            (root / 'item').write_text('original')
            script = pathlib.Path(__file__).resolve().parents[1] / 'scripts' / 'artifact_tree.py'
            first = subprocess.run([sys.executable, str(script), str(root)], capture_output=True, text=True, check=True)
            manifest = pathlib.Path(tmp) / 'manifest.json'
            manifest.write_text(first.stdout)
            self.assertEqual(subprocess.run([sys.executable, str(script), str(root), '--verify', str(manifest)], capture_output=True).returncode, 0)
            (root / 'item').write_text('changed')
            self.assertEqual(subprocess.run([sys.executable, str(script), str(root), '--verify', str(manifest)], capture_output=True).returncode, 1)

# Signed provenance envelope (experimental; official execution remains blocked)

`scripts/provenance.py` verifies an SSH signature over the exact bytes of a JSON manifest and checks SHA-256 digests of the artifact files it names. The allowed-signers file and expected principal are supplied independently by the caller; they must come from an operator-controlled trust policy, not from the manifest or an untrusted project checkout. The signature namespace is `herdr-ml-official-v1`. The implementation uses `ssh-keygen` from the caller's PATH, which must itself be trusted. An authorized key proves possession of that key; it is not independent scientific certification.

Schema version 1 requires `project`, `source_git_sha` (a full lowercase Git object ID), exact `argv`, integer `seed`, and an `artifacts` object with exactly these roles: `source`, `environment`, `dataset_manifest`, `dataset_content`, `train_split`, `validation_split`, `test_split`, `config`, `protocol`, `baseline`, and `review_evidence`. Each role contains `path` and lowercase SHA-256 `sha256`; paths are distinct, normalized relative file paths beneath the manifest directory. Use regular files; symlinks in any path component are rejected. The three split artifacts must have distinct content hashes; this catches exact duplicates but does not prove their contents are disjoint. Unknown top-level fields, duplicate JSON keys, invalid UTF-8 and nonstandard JSON constants are rejected. For a multi-file artifact, the producer can provide a deterministically produced archive, but a future execution layer must verify the extracted content and bind actual accesses to it.

Create and sign the exact manifest bytes with the owner's authorized signing key:

```sh
ssh-keygen -Y sign -n herdr-ml-official-v1 -f /path/to/owner-key /path/to/bundle/manifest.json
```

Call `verify_bundle(manifest, signature, allowed_signers, trusted_principal)` from a trusted Python process. Signing keys must not be committed or included in the bundle. Example trust policy format follows OpenSSH `allowed_signers` (for example, `owner ssh-ed25519 AAAA...`); its contents and principal selection are not authorized by the manifest itself. Review and protect policy and key ownership separately.

This verifier checks files at one point in time. It **does not** establish that the source archive actually corresponds to the claimed Git object, the dataset manifest describes the complete files used, the split files have the claimed semantics or no leakage, environment versions match the launched process, evidence is independently reviewed, authorization was granted for a particular actual execution, or files cannot change after verification. Source/data snapshot materialization, immutable access enforcement, authenticated operator policy provisioning, ownership fencing, execution-time binding and independent security/scientific review are required before official execution can be enabled. A signed statement is evidence of the signer's statement, not proof that the statement is scientifically correct.

## Diagnostic directory inventory

On Linux with procfs, `python3 scripts/artifact_tree.py /path/to/data > /path/outside/data-tree.json` inventories regular files by relative path, size and SHA-256, including empty directories. `python3 scripts/artifact_tree.py /path/to/data --verify /path/outside/data-tree.json` re-scans the actual files and exits nonzero on mismatch. The output file must be outside the scanned tree. The tool rejects symlinks and non-regular entries and detects several concurrent file/directory changes; the root's ancestor paths remain trusted.

This is a diagnostic inventory, not an atomic multi-file snapshot, a signature, a dataset license or leakage review, or a guarantee about files later used by a process. It is not connected to official execution. Store and independently authenticate the output; the official hold remains until an immutable execution boundary and authorization bind the actual accessed data and source to the run.
