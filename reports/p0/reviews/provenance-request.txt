#!/usr/bin/env python3
"""Verify a signed provenance envelope; this is not an execution authorizer.

The caller must supply an independently administered allowed-signers file and
principal. Verification does not lock artifacts or prove scientific validity.
Official execution remains disabled until a trusted execution boundary binds
verified immutable artifacts to the actual process and environment.
"""
import hashlib
import json
import os
from pathlib import Path, PurePosixPath
import re
import stat
import subprocess

NAMESPACE = 'herdr-ml-official-v1'
REQUIRED_ARTIFACTS = frozenset(('source', 'environment', 'dataset_manifest',
    'dataset_content', 'train_split', 'validation_split', 'test_split',
    'config', 'protocol', 'baseline', 'review_evidence'))


class ProvenanceError(Exception):
    pass


def _object(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ProvenanceError('duplicate JSON key')
        result[key] = value
    return result


def _hash_relative(root, name):
    if not isinstance(name, str) or not name or '\\' in name or '\x00' in name:
        raise ProvenanceError('invalid artifact path')
    path = PurePosixPath(name)
    if path.is_absolute() or any(part in ('', '.', '..') for part in name.split('/')):
        raise ProvenanceError('artifact path must be relative and normalized')
    directory = None
    file_fd = None
    try:
        directory = os.open(root, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        for component in path.parts[:-1]:
            next_dir = os.open(component, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory)
            os.close(directory)
            directory = next_dir
        file_fd = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=directory)
        before = os.fstat(file_fd)
        if not stat.S_ISREG(before.st_mode):
            raise ProvenanceError('artifact must be a regular file')
        digest = hashlib.sha256()
        with os.fdopen(file_fd, 'rb') as stream:
            file_fd = None
            for block in iter(lambda: stream.read(1024 * 1024), b''):
                digest.update(block)
            after = os.fstat(stream.fileno())
        if (before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns, before.st_ctime_ns) != (
                after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns, after.st_ctime_ns):
            raise ProvenanceError('artifact changed during verification')
        return digest.hexdigest()
    except OSError as exc:
        raise ProvenanceError('cannot safely read artifact') from exc
    finally:
        if file_fd is not None:
            os.close(file_fd)
        if directory is not None:
            os.close(directory)


def verify_bundle(manifest_path, signature_path, allowed_signers, principal):
    """Verify signed bytes and all named artifact bytes, returning parsed metadata.

    This is point-in-time checking, not a filesystem isolation or authorization
    boundary. Never use this as the sole precondition for launching a run.
    """
    if not isinstance(principal, str) or not principal or principal.startswith('-') or '\n' in principal:
        raise ProvenanceError('invalid trusted principal')
    try:
        manifest_path = Path(manifest_path)
        with manifest_path.open('rb') as handle:
            data = handle.read(1024 * 1024 + 1)
        if len(data) > 1024 * 1024:
            raise ProvenanceError('manifest too large')
        verified = subprocess.run(['ssh-keygen', '-Y', 'verify', '-f', str(allowed_signers),
            '-I', principal, '-n', NAMESPACE, '-s', str(signature_path)], input=data,
            stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=15, check=False)
        if verified.returncode:
            raise ProvenanceError('signature verification failed')
        manifest = json.loads(data, object_pairs_hook=_object)
    except (OSError, subprocess.SubprocessError, UnicodeError, json.JSONDecodeError) as exc:
        raise ProvenanceError('unable to verify signed manifest') from exc
    if not isinstance(manifest, dict) or type(manifest.get('schema_version')) is not int or manifest['schema_version'] != 1:
        raise ProvenanceError('invalid manifest schema')
    if not isinstance(manifest.get('project'), str) or not manifest['project'].strip():
        raise ProvenanceError('missing project')
    if not isinstance(manifest.get('source_git_sha'), str) or not re.fullmatch(r'[0-9a-f]{40}|[0-9a-f]{64}', manifest['source_git_sha']):
        raise ProvenanceError('invalid Git object ID')
    argv = manifest.get('argv')
    if not isinstance(argv, list) or not argv or not all(isinstance(arg, str) and '\x00' not in arg for arg in argv) or not argv[0]:
        raise ProvenanceError('invalid command argv')
    if type(manifest.get('seed')) is not int:
        raise ProvenanceError('invalid seed')
    artifacts = manifest.get('artifacts')
    if not isinstance(artifacts, dict) or set(artifacts) != REQUIRED_ARTIFACTS:
        raise ProvenanceError('missing or unsupported artifact roles')
    paths = set()
    for role, entry in artifacts.items():
        if not isinstance(entry, dict) or set(entry) != {'path', 'sha256'}:
            raise ProvenanceError('invalid artifact record: ' + role)
        name, expected = entry['path'], entry['sha256']
        if not isinstance(expected, str) or not re.fullmatch(r'[0-9a-f]{64}', expected):
            raise ProvenanceError('invalid artifact hash: ' + role)
        if not isinstance(name, str) or name in paths:
            raise ProvenanceError('duplicate or invalid artifact path')
        paths.add(name)
        if _hash_relative(manifest_path.parent, name) != expected:
            raise ProvenanceError('artifact digest mismatch: ' + role)
    return manifest

"""Adversarial CPU tests for signed provenance envelopes."""
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import subprocess
import tempfile
import unittest

MODULE = Path(__file__).resolve().parents[1] / 'scripts' / 'provenance.py'
if MODULE.exists():
    spec = importlib.util.spec_from_file_location('provenance', MODULE)
    p = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(p)
else:
    p = None


class SignedProvenanceTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name)
        self.bundle = self.root / 'bundle'
        self.bundle.mkdir()
        self.key = self.root / 'reviewer'
        subprocess.run(['ssh-keygen', '-q', '-t', 'ed25519', '-N', '', '-f', str(self.key)], check=True)
        self.allowed = self.root / 'allowed_signers'
        self.allowed.write_text('reviewer ' + self.key.with_suffix('.pub').read_text())
        self.artifacts = {}
        for role in ('source', 'environment', 'dataset_manifest', 'dataset_content', 'train_split',
                     'validation_split', 'test_split', 'config', 'protocol', 'baseline', 'review_evidence'):
            f = self.bundle / (role + '.dat')
            f.write_text('fixture ' + role)
            self.artifacts[role] = {'path': f.name, 'sha256': hashlib.sha256(f.read_bytes()).hexdigest()}
        self.manifest = {'schema_version': 1, 'project': 'demo', 'source_git_sha': 'a' * 40,
                         'argv': ['python', 'train.py'], 'seed': 42, 'artifacts': self.artifacts}
        self.path = self.bundle / 'manifest.json'
        self.write_signed()

    def write_signed(self):
        self.path.write_text(json.dumps(self.manifest, sort_keys=True, separators=(',', ':')) + '\n')
        signature = Path(str(self.path) + '.sig')
        if signature.exists(): signature.unlink()
        subprocess.run(['ssh-keygen', '-Y', 'sign', '-q', '-n', 'herdr-ml-official-v1', '-f', str(self.key), str(self.path)], check=True)

    def verify(self):
        return p.verify_bundle(self.path, Path(str(self.path) + '.sig'), self.allowed, 'reviewer')

    def test_real_signature_and_all_artifact_hashes(self):
        self.assertEqual(self.verify()['project'], 'demo')

    def test_changed_dataset_split_evidence_or_manifest_rejected(self):
        for role in ('dataset_content', 'test_split', 'review_evidence'):
            with self.subTest(role=role):
                f = self.bundle / self.artifacts[role]['path']
                original = f.read_bytes()
                f.write_bytes(original + b'changed')
                with self.assertRaises(p.ProvenanceError): self.verify()
                f.write_bytes(original)
        self.path.write_text(self.path.read_text().replace('demo', 'evil'))
        with self.assertRaises(p.ProvenanceError): self.verify()

    def test_untrusted_principal_and_unsigned_changes_rejected(self):
        with self.assertRaises(p.ProvenanceError):
            p.verify_bundle(self.path, Path(str(self.path) + '.sig'), self.allowed, 'attacker')
        self.manifest['argv'].append('--changed')
        self.path.write_text(json.dumps(self.manifest))
        with self.assertRaises(p.ProvenanceError): self.verify()

    def test_symlink_escape_and_missing_required_role_rejected(self):
        target = self.bundle / self.artifacts['test_split']['path']
        target.unlink()
        target.symlink_to(self.root / 'outside')
        with self.assertRaises(p.ProvenanceError): self.verify()
        target.unlink()
        del self.manifest['artifacts']['test_split']
        self.write_signed()
        with self.assertRaises(p.ProvenanceError): self.verify()

    def test_signed_path_traversal_and_duplicate_keys_rejected(self):
        self.manifest['artifacts']['test_split']['path'] = '../outside'
        self.write_signed()
        with self.assertRaises(p.ProvenanceError): self.verify()
        self.path.write_text('{"schema_version":1,"schema_version":1}')
        signature = Path(str(self.path) + '.sig')
        signature.unlink()
        subprocess.run(['ssh-keygen', '-Y', 'sign', '-q', '-n', 'herdr-ml-official-v1', '-f', str(self.key), str(self.path)], check=True)
        with self.assertRaises(p.ProvenanceError): self.verify()

    def test_unsigned_environment_or_config_changes_rejected(self):
        for role in ('source', 'environment', 'dataset_manifest', 'train_split', 'validation_split', 'config', 'protocol', 'baseline'):
            with self.subTest(role=role):
                artifact = self.bundle / self.artifacts[role]['path']
                before = artifact.read_bytes()
                artifact.write_bytes(before + b' altered')
                with self.assertRaises(p.ProvenanceError): self.verify()
                artifact.write_bytes(before)

# Signed provenance envelope (experimental; official execution remains blocked)

`scripts/provenance.py` verifies an SSH signature over the exact bytes of a JSON manifest and checks SHA-256 digests of the artifact files it names. The allowed-signers file and expected principal are supplied independently by the caller; they must come from an operator-controlled trust policy, not from the manifest or an untrusted project checkout. The signature namespace is `herdr-ml-official-v1`. The implementation uses `ssh-keygen` from the caller's PATH, which must itself be trusted. An authorized key proves possession of that key; it is not independent scientific certification.

Schema version 1 requires `project`, `source_git_sha` (a full lowercase Git object ID), exact `argv`, integer `seed`, and an `artifacts` object with exactly these roles: `source`, `environment`, `dataset_manifest`, `dataset_content`, `train_split`, `validation_split`, `test_split`, `config`, `protocol`, `baseline`, and `review_evidence`. Each role contains `path` and lowercase SHA-256 `sha256`; paths are distinct, normalized relative file paths beneath the manifest directory. Use regular files; symlinks in any path component are rejected. For a multi-file artifact, the producer can provide a deterministically produced archive, but a future execution layer must verify the extracted content and bind actual accesses to it.

Create and sign the exact manifest bytes with the owner's authorized signing key:

```sh
ssh-keygen -Y sign -n herdr-ml-official-v1 -f /path/to/owner-key /path/to/bundle/manifest.json
```

Call `verify_bundle(manifest, signature, allowed_signers, trusted_principal)` from a trusted Python process. Signing keys must not be committed or included in the bundle. Example trust policy format follows OpenSSH `allowed_signers` (for example, `owner ssh-ed25519 AAAA...`); its contents and principal selection are not authorized by the manifest itself. Review and protect policy and key ownership separately.

This verifier checks files at one point in time. It **does not** establish that the source archive actually corresponds to the claimed Git object, the dataset manifest describes the complete files used, the split files have the claimed semantics or no leakage, environment versions match the launched process, evidence is independently reviewed, authorization was granted for a particular actual execution, or files cannot change after verification. Source/data snapshot materialization, immutable access enforcement, authenticated operator policy provisioning, ownership fencing, execution-time binding and independent security/scientific review are required before official execution can be enabled. A signed statement is evidence of the signer's statement, not proof that the statement is scientifically correct.
